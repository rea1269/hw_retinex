\begin{frame}{问题与任务}
  \textbf{背景：}低光照图像存在亮度不足、噪声放大、细节丢失等问题，增强算法需兼顾提亮与保真。

  \vspace{0.6em}
  \textbf{本作业工作：}
  \begin{enumerate}
    \item \textbf{复现} RetinexFormer，在 5 个公开基准数据集上验证
    \item \textbf{自采验证}：两类场景（极限暗光纹理 / 高反差复杂光源），DNG/JPG 实拍数据
    \item \textbf{改进}：传统 baseline 对比 + 针对自采场景的后训练计划
  \end{enumerate}
\end{frame}

% ------------------------------------------------------------------ 3. Retinex 理论
\begin{frame}{Retinex 理论}
  \begin{columns}[T]
    \column{0.52\textwidth}
    \textbf{核心分解：}
    \[
      I(x,y) = R(x,y) \odot L(x,y)
    \]
    \begin{itemize}
      \item $I$：观测图像
      \item $L$：光照（Illumination），缓慢变化
      \item $R$：反射率（Reflectance），承载纹理与颜色
    \end{itemize}
    \vspace{0.5em}
    \small 思路：先估计并校正光照，再恢复反射细节——RetinexFormer 将此过程神经网络化。

    \column{0.46\textwidth}
    \centering
    \begin{tikzpicture}[
      box/.style={draw, rounded corners, minimum width=2.6cm, minimum height=0.75cm, align=center, font=\small},
      arr/.style={-{Stealth[length=2mm]}, thick}
    ]
      \node[box, fill=blue!8] (I) at (0,2) {低光输入 $I$};
      \node[box, fill=yellow!15] (L) at (0,0.8) {光照 $L$};
      \node[box, fill=green!10] (R) at (0,-0.4) {反射 $R$};
      \node[box, fill=orange!12] (O) at (0,-1.8) {增强输出};
      \draw[arr] (I) -- (L) node[midway, right, font=\scriptsize] {估计};
      \draw[arr] (I) -- (R) node[midway, left, font=\scriptsize] {分解};
      \draw[arr] (L) -- (O);
      \draw[arr] (R) -- (O);
    \end{tikzpicture}
  \end{columns}
\end{frame}

    \column{0.40\textwidth}
    \small
    \textbf{Self-ensemble 策略}
    \begin{itemize}
      \item 测试时对输入做 8 种几何变换（翻转 / 旋转组合）
      \item 分别推理后逆变换，对 8 路输出取平均
      \item 以约 $8\times$ 推理时间换取更稳定结果
    \end{itemize}
    \vspace{0.3em}
    \scriptsize
    \textbf{提升小结：}LOL 系列 SSIM +0.005$\sim$0.01；\\
    高分辨率集收益较小，但无明显负效应
  \end{columns}

% ------------------------------------------------------------------ 7. 分析：Domain Gap 与传统方法对比
\begin{frame}{分析：Domain Gap 与传统方法对比}
  \small
  \textbf{Domain Gap：}同一 mydata，不同 checkpoint 指标差异极大
  \begin{itemize}
    \item FiveK 权重：SSIM $\approx$ 0.21 \quad vs \quad SDSD\_outdoor：SSIM $\approx$ 0.65
    \item LOL\_v1 权重：PSNR $\approx$ 15.8 dB（公开集 25+ dB）$\Rightarrow$ 泛化不足
  \end{itemize}

  \vspace{0.3em}
  \scriptsize
  \renewcommand{\arraystretch}{1.12}
  \begin{tabular}{p{1.6cm}p{2.4cm}p{2.6cm}p{3.8cm}}
    \toprule
    算法 & 核心思想 & 优点 & 面对 LOL 的痛点 \\
    \midrule
    Gamma &
    全局幂函数映射 &
    极快，整体提亮明显 &
    亮部过曝，暗部噪点放大 \\
    CLAHE &
    分块直方图均衡 + clip\_limit &
    局部细节好，抑制噪声 &
    极暗场景提亮幅度有限 \\
    Gamma{+}CLAHE &
    全局拉伸 + HSV 润色 &
    兼顾亮度与纹理 &
    无法消除红绿彩噪 \\
    \bottomrule
  \end{tabular}

  \vspace{0.35em}
  \small
  \textbf{结论：}传统方法（\texttt{Enhancement/traditional\_baseline.py}）可作 baseline，\\
  彩噪与感知质量瓶颈仍需 RetinexFormer 等深度模型。
\end{frame}

    \begin{itemize}

    \begin{itemize}
      \item \textbf{冻结} \texttt{self.estimator}

      \item 解析权重 name 选择性冻结参数
      \item 训练入口：\texttt{image\_restoration\_model.py}
    \end{itemize}

    \vspace{0.2em}
    \textbf{预期目标：}分别在两类场景 fine-tune，\\
    缩小 domain gap，提升 mydata PSNR / SSIM
